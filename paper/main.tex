\begin{abstract}
Mechanistic interpretability seeks to understand large language models by identifying the causal pathways responsible for their behavior. Circuit tracing methods produce \emph{attribution graphs} encoding directed influence among model components—features, attention heads, and token positions—but these graphs routinely contain thousands of nodes and millions of edges, making manual inspection impractical. We propose \textbf{ASG} (Automated Summarization of attribution Graphs), a two-phase pipeline that (i) prunes the attribution graph by retaining nodes and edges that jointly score highly under both influence and relevance criteria, and (ii) clusters the retained feature nodes into semantically coherent \emph{supernodes} while enforcing a DAG-integrity constraint. We formalize the problem, describe the algorithm in detail, and evaluate it on synthetic multihop reasoning prompts. Preliminary results indicate that ASG substantially reduces graph size while preserving the bulk of attribution mass and producing structurally sound supernode graphs.
\end{abstract}

\section{Introduction}

The internal workings of large language models remain poorly understood. Mechanistic interpretability has emerged as a key approach for improving the transparency, reliability, and controllability of modern LLMs. Circuit tracing methods aim to identify the causal pathways underlying specific model behaviors by computing attribution among components such as attention heads, MLP neurons, residual streams, and token positions.

Sparse autoencoders (SAEs) have become a central tool for decomposing model activations into smaller, more interpretable units called \emph{features}, partially mitigating the polysemanticity of raw neurons \cite{elhage2022toy}. Variants such as transcoders and cross-layer transcoders connect these features across layers and enable the construction of full attribution graphs \cite{circuit-tracer}. However, the resulting graphs are often extremely complex, containing thousands of nodes and millions of edges, and their interpretation still requires substantial manual effort.

We introduce \textbf{ASG}, a method for automatically summarizing attribution graphs into compact, human-readable supernode graphs. The core idea is a two-phase pipeline: first, extract a key subgraph containing the most influential and relevant nodes and edges; second, aggregate functionally similar nodes into \emph{supernodes} to produce a compact summary. ASG targets a practical bottleneck in mechanistic interpretability research—the translation from raw attribution graphs to interpretable circuit diagrams—by replacing ad-hoc threshold selection and manual editing with a principled, reproducible algorithm.

\section{Related Work}

\subsection{Interpretability for Transformers}
A broad range of methods have been proposed to make transformer behavior more transparent. Saliency-based approaches \cite{simonyan2013deep,sundararajan2017axiomatic} highlight input tokens that most affect the output, but do not trace computation through internal layers. Attention visualization \cite{clark2019does} provides layer-wise snapshots but conflates correlation with causation. Probing classifiers \cite{belinkov2022probing} assess what information is encoded in representations without explaining how it is used. Causal intervention methods—activation patching \cite{meng2022locating} and path patching \cite{goldowsky2023localizing}—provide stronger causal evidence but scale poorly to full-graph analysis.

\subsection{Mechanistic Interpretability and Circuit Tracing}
Mechanistic interpretability seeks algorithmic explanations of model behavior \cite{olah2020zoom}. Superposition and polysemanticity in MLP neurons motivated sparse dictionary learning methods \cite{bricken2023monosemanticity,cunningham2023sparse}, and sparse autoencoders have since become the standard decomposition tool. Circuit tracing work has demonstrated useful findings on tasks such as indirect object identification \cite{wang2022interpretability} and greater-than comparison \cite{hanna2023does}, but these studies still rely on hand-curated subgraphs. The circuit-tracer framework \cite{circuit-tracer} automates attribution computation using cross-layer transcoders, producing full attribution graphs over features \cite{arxiv:2410.13032}. Our work addresses the downstream summarization step: converting these large graphs into interpretable supernode structures.

\section{Problem Formulation}

\subsection{Subgraph Extraction}
Let $G=(V,E,W)$ be a directed acyclic attribution graph, where the vertex set $V$ contains embedding nodes, feature nodes, and logit nodes, the edge set $E$ represents linear causal relationships between them, and $W \in \R^{N \times N}$ is the matrix of edge weights. We extract a subgraph
\[
G'=(V',E',W'), \qquad V'\subseteq V,\; E'\subseteq E,
\]
where $V' = V'_{\mathrm{emb}} \cup V'_{\mathrm{feat}} \cup V'_{\mathrm{logit}}$ partitions nodes into embedding, feature, and logit nodes.

We write $N = |V|$, $N' = |V'|$, and $N_{\mathrm{feat}} = |V'_{\mathrm{feat}}|$.

\subsection{Supernode Abstraction}
\label{sec:sn-def}

\begin{definition}[Supernode Mapping]
  A \emph{supernode mapping} is a surjection $\varphi: V' \to \{S_1, \ldots, S_K\}$ that assigns each node to one of $K$ supernodes, subject to:
  \begin{enumerate}[label=(\roman*)]
    \item $\varphi$ fixes embedding and logit nodes: each $v \in V'_{\mathrm{emb}} \cup V'_{\mathrm{logit}}$ forms its own singleton supernode.
    \item Each feature supernode $S_k$ has bounded layer span: $\max_{i \in S_k} \mathrm{layer}(i) - \min_{i \in S_k} \mathrm{layer}(i) \le \Delta$ for a fixed parameter $\Delta \ge 0$.
  \end{enumerate}
\end{definition}

The induced supernode graph $G^{\mathrm{SN}}$ has vertex set $\{S_1, \ldots, S_K\}$ and edges defined as follows.

\begin{definition}[Supernode Edge]
\label{def:sn-edge}
  For supernodes $S_k$ and $S_l$, let
  \[
    M(S_k \to S_l) \;=\; \sum_{i \in S_k,\, j \in S_l} W'_{ij}
  \]
  be the total signed attribution flowing from $S_k$ to $S_l$. The supernode edge weight is
\[
\begin{aligned}
M_\eta(C)
  &= \Bigl\{u : C_u \ge C_{(t)}\Bigr\}, \\
t
  &= \min\Biggl\{k :
      \frac{\sum_{j\le k} C_{(j)}}{\sum_j C_{(j)}} \ge \eta
    \Biggr\},
\end{aligned}
\]
  For each pair $\{S_k, S_l\}$, only the direction with strictly greater absolute mass is retained; ties (including both-zero) produce no edge. The edge set of $G^{\mathrm{SN}}$ is $\bigl\{(S_k \to S_l) : W^{\mathrm{SN}}(S_k \to S_l) \neq 0\bigr\}$.
\end{definition}

\subsection{Desiderata}
\label{sec:desiderata}

\textbf{(D1) DAG integrity.} The induced supernode graph must remain acyclic.
\textbf{(D2) Mechanism Preservation.} The supernode graph should preserve the causal flow of the original graph.
\textbf{(D3) Concept coherence.} Nodes in the same supernode should play similar structural roles in the circuit.
\textbf{(D4) Cluster balance.} The supernode graph should be balanced, i.e., the number of nodes in each supernode should be roughly equal.
\textbf{(D5) Minimality.} The supernode graph should be as small as possible.
\section{ASG: Automated Summarization of Attribution Graphs}

\paragraph{Step 1: Subgraph Extraction.}
For each attribution graph $G$, ASG constructs a candidate subgraph $G'$ by retaining nodes and edges that score highly under a \emph{combined} criterion that fuses two complementary signals: \textbf{influence}---the indirect effect of an internal node on the target logits \cite{circuit-tracer,arxiv:2410.13032}---and \textbf{relevance}---the indirect dependence of an internal node on the input tokens. Influence is propagated backward from logits through input-normalized edges; relevance is propagated forward from embeddings through output-normalized edges.

\paragraph{Influence and relevance scores.}
Let $W_{ij}$ denote the edge weight from node $i$ to node $j$ in $G'$. We define input- and output-normalized matrices
\[
W^{\mathrm{in}}_{ij} \,=\, \frac{|W_{ij}|}{\sum_{k}|W_{kj}|}, \qquad
W^{\mathrm{out}}_{ij} \,=\, \frac{|W_{ij}|}{\sum_{k}|W_{ik}|},
\]
so that each column of $W^{\mathrm{in}}$ sums to 1 (each target node's incoming mass sums to 1) and each row of $W^{\mathrm{out}}$ sums to 1 (each source node's outgoing mass sums to 1). Let $\ell\in\R^n$ be the logit-weight vector and $\tau\in\R^n$ the token-weight vector (set to normalized SHAP values \cite{shap}). Node scores are computed via the Neumann series, which converges because $G'$ is a DAG:
\[
\mathrm{Inf}_u = \bigl(\ell^\top (I-(W^{\mathrm{in}})^{\top})^{-1}\bigr)_u, \qquad
\mathrm{Rel}_u = \bigl(\tau^\top (I-W^{\mathrm{out}})^{-1}\bigr)_u.
\]
Edge scores decompose node scores by edge fraction:
\[
\mathrm{Inf}_{uv} = \mathrm{Inf}_v\,W^{\mathrm{in}}_{uv}, \qquad
\mathrm{Rel}_{uv} = \mathrm{Rel}_u\,W^{\mathrm{out}}_{uv},
\]
where $uv$ denotes the edge from source $u$ to target $v$, so $W^{\mathrm{in}}_{uv}$ is the fraction of $v$'s input mass from $u$, and $W^{\mathrm{out}}_{uv}$ is the fraction of $u$'s output mass to $v$.

\paragraph{Combined score.}
Because influence and relevance live on different scales and are heavily zero-inflated, we normalize each to $[0,1]$ via the rank-percentile transform
\[
\mathrm{rank}(s)_u \;=\; \frac{|\{v : s_v < s_u\}|}{N - 1},
\]
and combine them as the weighted geometric mean
\[
C_u \;=\; \bigl(\widehat{\mathrm{Inf}}_u + \varepsilon\bigr)^{\alpha}\,\bigl(\widehat{\mathrm{Rel}}_u + \varepsilon\bigr)^{1-\alpha}, \quad \alpha\in[0,1].
\]
The geometric form penalizes nodes weak on either axis, encoding the desideratum that a retained node must be \emph{both} reachable from input tokens and influential on the output. Arithmetic and harmonic means are also supported.

\paragraph{Cumulative-mass thresholding.}
Given a combined score vector $C$, we sort entries in descending order and keep the smallest prefix whose cumulative mass exceeds a threshold $\eta\in[0,1]$:
\[
M_\eta(C) \;=\; \Bigl\{u : C_u \ge C_{(t)}\Bigr\}, \qquad
t = \min\Bigl\{k : \tfrac{\sum_{j\le k} C_{(j)}}{\sum_j C_{(j)}} \ge \eta\Bigr\},
\]
where $C_{(j)}$ denotes the $j$-th largest score. This is the same cumulative-mass rule used by prior circuit tracing work \cite{circuit-tracer}, applied here to the combined ranking rather than to a single metric.

We apply this rule at the node level with threshold $\eta_V$ to obtain a node mask, recompute edge-level combined scores $C_{uv}$ on the node-pruned graph, then apply it again at the edge level with threshold $\eta_E$. A final cleanup step iteratively removes dangling interior nodes (no surviving incoming or outgoing edges). The full procedure is given in \Cref{alg:prune}. We found that a threshold $\eta_V$ of 0.02 is sufficient of selecting important nodes for our clustering step.

\paragraph{Step 2: Supernode Clustering.}
After pruning, the graph typically still contains groups of features with similar mechanistic roles—a consequence of feature splitting \cite{split} and the structure of the underlying feature manifold. 
We apply a version of \textbf{pattern-based clustering}, or \textbf{citation-based clustering} classified in \cite{clustering-survey} to group features into supernodes to produce a compact summary.

We define two similarity matrices based on shared neighborhood structure weighted by attribution scores:
\[
S_{\mathrm{out}} = \operatorname{ReLU}\!\left(\operatorname{CosNorm}(W\,\operatorname{diag}(\mathrm{Inf})\,W^{\top})\right),
\]
\[
S_{\mathrm{in}} = \operatorname{ReLU}\!\left(\operatorname{CosNorm}(W^{\top}\operatorname{diag}(\mathrm{Rel})\,W)\right),
\]
where $W \in \R^{N_\mathrm{feat} \times N_\mathrm{feat}}$ is the source-to-target weight matrix (rows index source nodes, columns index target nodes), $\mathrm{Inf}, \mathrm{Rel} \in \R^n$ are the node influence and relevance vectors, and $\operatorname{CosNorm}$ applies cosine normalization. $S_\mathrm{out}$ measures shared downstream targets weighted by their influence; $S_\mathrm{in}$ measures shared upstream sources weighted by their relevance. We combine them as
\[
S = F_{\mathrm{mean}}\bigl(S_{\mathrm{out}}, S_{\mathrm{in}}\bigr),
\]
where $F_{\mathrm{mean}}$ denotes arithmetic, harmonic, or geometric mean (a hyperparameter).

To penalize pairings whose merged layer span would threaten DAG integrity, we apply a layer-span decay:
\[
P_{ij} = e^{-\lambda \Delta_{ij}},
\]
where $\lambda > 0$ and $\Delta_{ij}$ is the layer difference between nodes $i$ and $j$. The final affinity matrix is
\[
\tilde{S}_{ij} = S_{ij} \cdot P_{ij} \in [0,1]^{N_{\mathrm{feat}} \times N_{\mathrm{feat}}}.
\]

Since $\tilde{S}$ is symmetric and non-negative, we apply spectral clustering with $\tilde{S}$ as the affinity matrix. This embeds feature nodes using the eigenvectors of the graph Laplacian and partitions them with K-means to obtain $K$ supernodes. In practice, we clamp $\tilde{S}$ with a small epsilon to ensure that the graph Laplacian remains connected.

\paragraph{DAG enforcement.}
After spectral clustering, we enforce DAG integrity (D2) by iteratively resolving cycles in the cluster digraph. Supernode edges are computed via \Cref{def:sn-edge}: for each cluster pair, only the direction with strictly greater absolute inter-cluster mass is retained, eliminating all 2-cycles by construction. We then identify strongly connected components (SCCs) of size $\geq 2$; within each SCC, the cluster with the widest layer span is split at its median layer. This repeats until no SCCs remain.

\section{Experiments}

We evaluate ASG on two synthetic datasets:
\begin{itemize}
  \item \textbf{Multihop reasoning prompts:} factual recall queries that require two steps of reasoning, similar to \cite{circuit-tracer}.
  \item \textbf{Analogies:} prompts of the format ``The saying goes: A is to B as C is to,'' testing whether the language model can understand the relation between A and B and apply it to C.
\end{itemize}

\subsection{Supernode graph Evaluation}
\label{sec:cluster-eval}

We evaluate ASG against three $K$-matched baselines on $N{=}55$ pruned attribution graphs from the analogies dataset, with $n$ denoting the number of non-fixed feature nodes in each graph.

\paragraph{Baselines.}
All baselines operate on the same pruned feature set as ASG and are forced to produce the same number of supernodes $K$.
\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=2pt]
  \item \textbf{Modularity.} Greedy modularity maximization on the symmetrized weighted adjacency, constrained to $K$ communities via the \texttt{best\_n} parameter of \texttt{networkx.community.greedy\_modularity\_communities} \cite{clauset2004finding}. Captures connectivity-based community structure but is blind to edge direction.
  \item \textbf{Spectral-RBF.} Spectral clustering with an RBF kernel over per-feature bidirectional adjacency rows (concatenated incoming and outgoing weights). Recovers dense local geometry without using the attribution-weighted similarity defined in Step~2 of ASG.
  \item \textbf{K-means.} K-means on the same bidirectional adjacency features. Simplest non-trivial partition baseline.
\end{itemize}

\paragraph{Scoring metrics.}
We report six metrics, none of which use the affinity matrix $\tilde{S}$ that drives ASG clustering, grouped by the desiderata of \Cref{sec:desiderata}.
\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=2pt]
  \item \textbf{Back-edge fraction} (D1, $\downarrow$): share of supernode-graph edge mass flowing against layer order,
  $\frac{\sum_{i\to j:\,\mathrm{layer}(j)<\mathrm{layer}(i)} |W^{\mathrm{SN}}_{ij}|}{\sum_{i\to j}|W^{\mathrm{SN}}_{ij}|}$, with $W^{\mathrm{SN}}$ from \Cref{def:sn-edge}.
  \item \textbf{Internal independence} (D2/D3, $\uparrow$): per-node external-mass ratio averaged over $V'$,
  $\tfrac{1}{|V'|}\sum_u \tfrac{|W^{\mathrm{ext}}_u|}{|W^{\mathrm{ext}}_u|+|W^{\mathrm{int}}_u|}$,
  where internal/external is taken with respect to $u$'s supernode.
  \item \textbf{Cluster-size CV} (D4, $\downarrow$): coefficient of variation of supernode sizes.
  \item \textbf{Size entropy} (D4, $\uparrow$): normalized entropy $H(p)/\log K \in [0,1]$ of the size distribution; 1 = uniform.
  \item \textbf{Opposing-sign fraction} (D3, $\downarrow$): fraction of intra-supernode pairs whose net signed contributions to logit nodes have opposite signs.
  \item \textbf{Output-direction cosine gap} (D3, $\uparrow$): mean intra-supernode pairwise cosine of feature output vectors minus the cross-supernode baseline, where a feature's output vector is its column of $W'$ restricted to logit-node rows. Uses only the output-to-logit slice of the adjacency, so it is independent of the bidirectional edge profile that drives clustering.
\end{itemize}

\paragraph{Results.}
\Cref{tab:cluster-metrics} summarizes performance. The three ASG variants together dominate all causal-coherence metrics: at every $K$, ASG-arith records 3--7$\times$ fewer back-edges than the best baseline (D1) and ranks first on internal independence, while ASG-harm and ASG-geo essentially eliminate opposing-sign pairs (0.008--0.086 vs.\ 0.235--0.400 for baselines) and push intra-supernode output directions into near-unit alignment (cosine gap 0.96--0.99 vs.\ $-0.04$ to 0.44, D3). The mean operator has a clear interpretation: arithmetic preserves more in/out structural signal and so wins the structural metrics (back-edge, independence), while the harmonic and geometric means sharpen the similarity by penalizing pairs weak on either side, producing tighter directionally-consistent clusters but slightly more back-edges. Modularity wins size entropy (D4) by virtue of its balance-favoring objective, but loses every causal-coherence metric---a direct consequence of being undirected. Internal independence saturates above 0.90 across all methods and is not discriminative on this dataset; we retain it for completeness but caution against treating it as a primary signal. A full per-$\lambda$ ablation is deferred to the ablation section.

\subsection{Steering Case Study}
\label{sec:steering}

The metrics of \Cref{sec:cluster-eval} are graph-theoretic: they evaluate the supernode partition through the lens of the (already-computed) attribution graph. To verify that the partition is also \emph{causally} meaningful, we perform two ablation experiments via the cross-layer transcoder substrate. For each feature, we ablate its post-encoder activation by clamping it to zero at its original $(\text{layer}, \text{position})$ index and propagate the modified state through the rest of the model.

\paragraph{Setup.}
For a given prompt and pruned graph $G'$, let $G^{\mathrm{SN}}$ be a supernode graph produced by one of five methods: ASG-spectral, ASG-agglomerative, modularity, spectral-RBF, or K-means (\Cref{sec:cluster-eval}). ASG-spectral selects its supernode count $K^\star$ automatically via the eigengap heuristic on $\tilde{S}$; all three baselines (and ASG-agglomerative, via the same auto-$K$ procedure) are forced to the same $K^\star$ so cross-method differences reflect the partition rather than its granularity. Let $z^0$ denote the original CLT activations on the prompt and $\ell^0 \in \R^{|\mathcal{V}|}$ the resulting last-position logit vector.

\paragraph{Exp~B: Causal edge knockout.}
For each feature-type supernode $S$, we replay the prompt with all CLT activations belonging to $S$ clamped to zero, yielding modified activations $z^{S}$. For every directed supernode edge $(S_A\!\to\!S_B)$ with weight $W^{\mathrm{SN}}(S_A\!\to\!S_B) > 0$, we record the downstream activation-drop ratio
\[
  r_{A \to B} \;=\; \frac{\bar z^{\,S_A}(S_B)}{\bar z^{\,0}(S_B) + \varepsilon},
\]
where $\bar z^{\,(\cdot)}(S_B)$ is the mean post-ablation activation across the CLT features in $S_B$ at their respective $(\text{layer}, \text{position})$ indices. We summarize each supernode graph by the Spearman rank correlation
\[
  \rho_{\text{edge}} \;=\; \mathrm{Spearman}\!\bigl(W^{\mathrm{SN}}(S_A\!\to\!S_B),\; 1 - r_{A \to B}\bigr)
\]
over all positive-weight edges. A causally faithful supernode graph should yield $\rho_{\text{edge}} > 0$: heavier supernode edges should imply larger downstream activation drops when their source is ablated. Crucially, $\rho_{\text{edge}}$ probes whether the \emph{supernode edges themselves}---defined by \Cref{def:sn-edge} as aggregated attribution mass---predict actual causal effects under intervention, which is the strongest test of mechanism preservation (D2) we can construct from the partition alone.

\paragraph{Exp~D: Per-feature logit-$\Delta$ coherence.}
For each feature $f$ inside a feature supernode $S$, we ablate $f$ in isolation and record the last-position logit perturbation $\Delta_f = \ell^{\,f} - \ell^{\,0} \in \R^{|\mathcal{V}|}$. Within $S$, we compute the mean pairwise cosine of these perturbation vectors,
\[
  \kappa_{\text{intra}}(S) \;=\; \frac{2}{|S|\,(|S|{-}1)} \sum_{\{f, g\}\subseteq S,\, f\neq g} \cos\bigl(\Delta_f, \Delta_g\bigr),
\]
and average $\kappa_{\text{intra}}(S)$ across feature supernodes. For the cross-supernode baseline $\kappa_{\text{inter}}$, we sample $M{=}1000$ random pairs of features drawn from \emph{different} supernodes and average their pairwise cosines. We report the gap $\kappa_{\text{intra}} - \kappa_{\text{inter}}$.

A high gap indicates that features grouped by the method push the logit distribution in genuinely similar causal directions, providing an interventional counterpart to the output-direction cosine of \Cref{sec:cluster-eval}: the latter measures alignment of \emph{attributed} output contributions (linear, from $W'$), while $\kappa_{\text{intra}} - \kappa_{\text{inter}}$ measures alignment of \emph{realized} logit effects under non-linear forward replay.

\paragraph{Compared methods and reporting.}
All five methods are evaluated on the same prompts with identical CLT-ablation interventions, so differences in $\rho_{\text{edge}}$ and $\kappa_{\text{intra}} - \kappa_{\text{inter}}$ are attributable to the partition. We report each metric averaged over graphs in the evaluation set.

\section{Limitations and Future Work}

ASG inherits limitations from the underlying attribution backend: attribution quality depends on transcoder fidelity and may be biased toward high-magnitude edges that are not causally primary. The cumulative-mass thresholds require some domain knowledge to tune. The DAG constraint is enforced via a soft layer-span penalty rather than a hard combinatorial guarantee, which may permit mild violations on very dense graphs.

Future work includes adaptive threshold selection based on graph statistics, task-conditioned scoring that weights tokens by task-specific importance, stronger causal validation via interventional experiments, and evaluation across model families beyond Gemma.

\section{Conclusion}

We present ASG, an automated pipeline for summarizing attribution graphs in circuit tracing workflows. By combining joint influence--relevance pruning with DAG-constrained supernode clustering, ASG produces compact summaries that preserve attribution mass while grouping features into semantically coherent units. This addresses a practical bottleneck in mechanistic interpretability: the gap between raw attribution graphs and human-interpretable circuit diagrams. We provide a formal problem formulation, a detailed algorithm, and an experimental protocol. We hope ASG will serve as a standardized, reproducible foundation for attribution graph analysis.

\section{Pruning Algorithm}
\label{sec:appendix-prune}

\begin{algorithm}[H]
\caption{Prune attribution graph (combined score)}
\label{alg:prune}
\begin{algorithmic}[1]
\Require Graph $G$, \texttt{token\_weights} $\tau$, \texttt{logit\_weights} $\ell$, thresholds $(\eta_V,\eta_E)$, mixing $\alpha$
\Ensure $(\texttt{node\_mask},\texttt{edge\_mask})$
\State Compute $\mathrm{Inf}_u = (\ell^\top (I-(W^{\mathrm{in}})^\top)^{-1})_u$ for all $u$
\State Compute $\mathrm{Rel}_u = (\tau^\top (I-W^{\mathrm{out}})^{-1})_u$ for all $u$
\State $\widehat{\mathrm{Inf}} \gets \mathrm{rank}(\mathrm{Inf})$;\quad $\widehat{\mathrm{Rel}} \gets \mathrm{rank}(\mathrm{Rel})$
\State $C^V_u \gets (\widehat{\mathrm{Inf}}_u+\varepsilon)^{\alpha}(\widehat{\mathrm{Rel}}_u+\varepsilon)^{1-\alpha}$
\State $\texttt{node\_mask} \gets M_{\eta_V}(C^V)$;\quad force-keep all embedding and logit nodes
\State Zero out rows/columns of $W'$ not in \texttt{node\_mask}
\State Compute $\mathrm{Inf}_{uv} = \mathrm{Inf}_v W^{\mathrm{in}}_{uv}$ and $\mathrm{Rel}_{uv} = \mathrm{Rel}_u W^{\mathrm{out}}_{uv}$ on pruned graph
\State $\widehat{\mathrm{Inf}}_{uv} \gets \mathrm{rank}(\mathrm{Inf}_{uv})$;\quad $\widehat{\mathrm{Rel}}_{uv} \gets \mathrm{rank}(\mathrm{Rel}_{uv})$
\State $C^E_{uv} \gets (\widehat{\mathrm{Inf}}_{uv}+\varepsilon)^{\alpha}(\widehat{\mathrm{Rel}}_{uv}+\varepsilon)^{1-\alpha}$
\State $\texttt{edge\_mask} \gets M_{\eta_E}(C^E)$
\State \textbf{repeat} remove interior nodes with no surviving in- or out-edges \textbf{until} fixed point
\State \Return $(\texttt{node\_mask},\texttt{edge\_mask})$
\end{algorithmic}
\end{algorithm}
